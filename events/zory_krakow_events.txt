# 1. Zryw krakowski — uruchamiany po nacjonalizmie i 40 szału kultury polskiej.
zory_krakow.1 = {
    type = country_event
    placement = root
    title = zory_krakow.1.t
    desc = zory_krakow.1.d
    flavor = zory_krakow.1.f
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    duration = 3

    immediate = { set_variable = zory_krakow_revolt_started }

    

    option = {
        name = zory_krakow.1.a
        default_option = yes
        # Tylko ziemie bezpośrednio należące do Austrii
        # i będące ojczyzną kultury polskiej.
        c:AUS = {
            every_scope_state = {
                limit = { state_region = { is_homeland = cu:polish } }
                set_state_owner = c:KRA
            }
        }
        c:POL ?= {
            trigger_event = { id = zory_krakow.2 popup = yes }
        }
        c:UKR ?= {
            trigger_event = { id = zory_krakow.3 popup = yes }
        }
        # Gra umożliwia dokonanie wyboru zanim Austria wystąpi z ultimatum.
        trigger_event = { id = zory_krakow.4 days = 30 popup = yes }
    }
    # Oba panstwa musza byc niepodlegle w momencie zaproszenia.
        if = {
            limit = {
                exists = c:POL
                exists = c:UKR
                c:POL = { is_subject = no }
                c:UKR = { is_subject = no }
            }
            c:POL = { trigger_event = { id = zory_krakow.2 popup = yes } }
            c:UKR = { trigger_event = { id = zory_krakow.3 popup = yes } }
        }
        # Gra umożliwia dokonanie wyboru zanim Austria wystąpi z ultimatum.
        trigger_event = { id = zory_krakow.4 days = 30 popup = yes }
    }

    option = {
        name = zory_krakow.1.b
        # Odmowa kończy łańcuch w tej rozgrywce.
    }

# 4. Ultimatum Wiednia: austriacki cel wojenny to aneksja Krakowa.
zory_krakow.4 = {
    type = country_event
    placement = root
    title = zory_krakow.4.t
    desc = zory_krakow.4.d
    flavor = zory_krakow.4.f
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    duration = 3

        option = {
        name = zory_krakow.4.a
        default_option = yes

        if = {
            limit = {
                exists = c:AUS
                exists = c:KRA
            }

            # Jesli Krakow jest poddanym,
            # najpierw uzyskuje niepodleglosc.
            c:KRA = {
                if = {
                    limit = {
                        is_subject = yes
                    }

                    make_independent = yes
                }
            }

            # Austria rozpoczyna konflikt
            # przeciw niepodleglemu Krakowowi.
            c:AUS = {
                set_variable = zory_krakow_austria_initiator

                create_diplomatic_play = {
                    type = dp_zory_krakow_annex
                    target_country = c:KRA
                    escalation = 0
                }
            }
        }
    }
}

# 5. Zwycięstwo: Polska Kongresowa już otrzymała Kraków i jego ziemie.

# 5. Zwyciestwo w wojnie z Austria: Polska Kongresowa anektuje Krakow.
# Event uruchamiany przez zory_krakow_resolve_after_war dla POL.
# Nie da sie przeprowadzic aneksji przez samo wywolanie eventu z konsoli:
# potrzebne sa DWA osobne znaczniki (po kapitulacji i po zakonczeniu wojny).
zory_krakow.5 = {
    type = country_event
    placement = root
    title = zory_krakow.5.t
    desc = zory_krakow.5.d
    flavor = zory_krakow.5.f
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    duration = 3

    # Wykonywane w chwili utworzenia wydarzenia, przed wyswietleniem okna.
    immediate = {
        if = {
            limit = {
                # Zmienna stworzona w on_war_end tylko po zwyciestwie Krakowa.
                has_variable = zory_krakow_victory_pending
                NOT = { has_variable = zory_krakow_reunification_done }
                exists = c:KRA
                exists = c:POL
                c:KRA = {
                    has_variable = zory_krakow_revolt_started
                    has_variable = zory_krakow_austria_defeated
                }
            }

            # ROOT to Polska Kongresowa (POL). Ustawiamy znacznik przed
            # aneksja, aby operacja byla jednorazowa.
            set_variable = zory_krakow_reunification_done
            remove_variable = zory_krakow_victory_pending
            annex = c:KRA
        }
    }

    option = {
        name = zory_krakow.5.a
        default_option = yes
    }
}
