# Żorska Szkoła Ekonomii — Kraków i powstanie galicyjskie
# Cel: sprawdzanie warunków, dołączenie sojuszników i rozstrzygnięcie wojny.

on_monthly_pulse = {
    on_actions = { zory_krakow_monthly_check }
}

zory_krakow_monthly_check = {
    effect = {
        if = {
            limit = {
                exists = c:KRA
                exists = c:AUS
                c:KRA = {
                    has_technology_researched = nationalism
                    NOT = { has_variable = zory_krakow_revolt_started }
                    is_at_war = no
                }
                c:AUS = {
                    is_at_war = no
                    any_scope_state = { state_region = { is_homeland = cu:polish } }
                }
                cu:polish = { culture_current_fervor >= 40 }
            }
            c:KRA = {
                trigger_event = { id = zory_krakow.1 popup = yes }
            }
        }
    }
}

# Dołączenie wspierających państw następuje przy otwarciu właściwego DP.
# Daje to Polsce i Ukrainie czas na odpowiedź przed jego wybuchem.
on_diplomatic_play_started = {
    on_actions = { zory_krakow_join_backers }
}

zory_krakow_join_backers = {
    effect = {
        if = {
            limit = {
                scope:initiator = { has_variable = zory_krakow_austria_initiator }
                scope:target = { has_variable = zory_krakow_revolt_started }
                exists = c:POL
                exists = c:UKR
                c:POL = { is_subject = no }
                c:UKR = { is_subject = no }
            }
            if = {
                limit = {
                    exists = c:POL
                    c:POL = { has_variable = zory_krakow_polish_support }
                }
                add_target_backers = { c:POL }
            }
            if = {
                limit = {
                    exists = c:UKR
                    c:UKR = { has_variable = zory_krakow_ukrainian_support }
                }
                add_target_backers = { c:UKR }
            }
        }
    }
}

# Kapitulacja: ROOT to kraj kapitulujacy; scope:diplomatic_play to wojna.
# Dlatego porazke Austrii zapisujemy wylacznie, gdy to sama Austria
# kapituluje w NASZYM typie posuniecia dyplomatycznego.
on_capitulation = {
    on_actions = { zory_krakow_record_austrian_defeat }
}

zory_krakow_record_austrian_defeat = {
    effect = {
        if = {
            limit = {
                # ROOT musi byc Austria; zmienna przypisywana jest tylko AUS
                # w zory_krakow.4 (przy rozpoczeciu ultimatum).
                has_variable = zory_krakow_austria_initiator
                exists = c:KRA
                c:KRA = { has_variable = zory_krakow_revolt_started }
                scope:diplomatic_play = {
                    is_diplomatic_play_type = dp_zory_krakow_annex
                }
            }
            c:KRA = { set_variable = zory_krakow_austria_defeated }
        }
    }
}

# Root on_war_end to diplomatic_play.
# scope:actor = inicjator (Austria), scope:target = cel (Krakow).
on_war_end = {
    on_actions = { zory_krakow_resolve_after_war }
}

zory_krakow_resolve_after_war = {
    effect = {
        if = {
            limit = {
                is_diplomatic_play_type = dp_zory_krakow_annex
                scope:actor = { has_variable = zory_krakow_austria_initiator }
                scope:target = { has_variable = zory_krakow_revolt_started }
            }

            # Warunek zwyciestwa: Austria rzeczywiscie skapitulowala.
            if = {
                limit = {
                    exists = c:POL
                    exists = c:KRA
                    c:KRA = {
                        has_variable = zory_krakow_revolt_started
                        has_variable = zory_krakow_austria_defeated
                    }
                }
                c:POL = {
                    if = {
                        limit = {
                            NOT = { has_variable = zory_krakow_reunification_done }
                        }
                        # Tylko ten znacznik uprawnia event .5 do aneksji.
                        set_variable = zory_krakow_victory_pending
                        trigger_event = { id = zory_krakow.5 popup = yes }
                    }
                }
            }
            else_if = {
                limit = {
                    exists = c:KRA
                    c:KRA = { has_variable = zory_krakow_revolt_started }
                }
                c:KRA = {
                    trigger_event = { id = zory_krakow.6 popup = yes }
                }
            }

            # Sprzatanie znacznikow deklaracji wsparcia.
            c:POL ?= { remove_variable = zory_krakow_polish_support }
            c:UKR ?= { remove_variable = zory_krakow_ukrainian_support }
            scope:actor = { remove_variable = zory_krakow_austria_initiator }
        }
    }
}
